cmake_minimum_required(VERSION 3.28)
project(StreamStrikeBrandingContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
# Original OBS resource from donor 72e86655715d9a25d69ce446aa4dd44ba58e10d2.
# Do not substitute a new drawing or link QtSvg directly to hide absent runtime plugins.
set(OBS_DONOR_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../foundation/obs-studio" CACHE PATH "Pinned original OBS source")
set(OBS_PLUS_SVG "${OBS_DONOR_SOURCE}/frontend/forms/images/plus.svg")
if(NOT EXISTS "${OBS_PLUS_SVG}")
  message(FATAL_ERROR "Check out the pinned OBS donor before building the original SVG contract")
endif()
file(SHA256 "${OBS_PLUS_SVG}" OBS_PLUS_SVG_SHA256)
if(NOT OBS_PLUS_SVG_SHA256 STREQUAL "559bbcc914d0ee4a4c61f2af615f4cfac4100f90f93ac2de0b8a9cf3a4af9891")
  message(FATAL_ERROR "Original OBS plus.svg does not match the pinned resource")
endif()
get_filename_component(OBS_PLUS_SVG "${OBS_PLUS_SVG}" ABSOLUTE)
string(REPLACE "&" "&amp;" OBS_PLUS_SVG_XML "${OBS_PLUS_SVG}")
string(REPLACE "<" "&lt;" OBS_PLUS_SVG_XML "${OBS_PLUS_SVG_XML}")
string(REPLACE ">" "&gt;" OBS_PLUS_SVG_XML "${OBS_PLUS_SVG_XML}")
set(OBS_SVG_QRC "${CMAKE_CURRENT_BINARY_DIR}/original-obs-svg.qrc")
file(WRITE "${OBS_SVG_QRC}" "<RCC><qresource prefix=\"/res\"><file alias=\"images/plus.svg\">${OBS_PLUS_SVG_XML}</file></qresource></RCC>\n")
add_executable(streamstrike-icon-contract icon_test.cpp master.qrc "${OBS_SVG_QRC}")
target_include_directories(streamstrike-icon-contract PRIVATE ../../foundation/branding)
target_link_libraries(streamstrike-icon-contract PRIVATE Qt6::Widgets)
